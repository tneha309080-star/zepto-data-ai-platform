Q7_CATEGORY_SUMMARY
============================================================
SELECT
                c.category_name,
                COUNT(b.book_id) AS book_count,
                ROUND(AVG(b.price_gbp), 2) AS average_price_gbp
            FROM categories AS c
            JOIN books AS b
                ON c.category_id = b.category_id
            GROUP BY c.category_id, c.category_name
            ORDER BY book_count DESC
